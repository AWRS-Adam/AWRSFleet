🗄 Supabase Setup

Run this SQL once in your Supabase dashboard → SQL Editor to create the required tables:

create table if not exists fi_settings (key text primary key, value jsonb);
create table if not exists fi_inspections (id text primary key, data jsonb, created_at timestamptz default now());
create table if not exists fi_defect_log (log_id text primary key, data jsonb);
create table if not exists fi_maint_log (unit_id text, item_id text, data jsonb, primary key(unit_id,item_id));
create table if not exists fi_odometer (unit_id text primary key, data jsonb);
alter table fi_settings enable row level security;
alter table fi_inspections enable row level security;
alter table fi_defect_log enable row level security;
alter table fi_maint_log enable row level security;
alter table fi_odometer enable row level security;
create policy "anon all" on fi_settings for all using (true) with check (true);
create policy "anon all" on fi_inspections for all using (true) with check (true);
create policy "anon all" on fi_defect_log for all using (true) with check (true);
create policy "anon all" on fi_maint_log for all using (true) with check (true);
create policy "anon all" on fi_odometer for all using (true) with check (true);
